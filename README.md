# LlamaParse_to_parse_PDF and use Agentic AI to ask questions in Natural Language. ( JavaScript and Python )

In this repo - I would like to show how to use LlamaParse to parse a PDF document.

The Input document to be converted is a PDF file ( e.g: Electoral Roll PDF listing all the voters in a certain constituency).

Each page is in this Voters List is an image of voters with their voter id, name, age, gender address etc, and contents cannot be extracted directly. A Sample of the page is shown below.

These are about 20 to 30 pages of PNG or JPG images, which is now showing up as a PDF document. The challenge is that the 

The text in this cannot be extracted directly, **Each page has to be converted from the IMG ( PNG or JPG ) and the text has to be 
extracted using OCR, and this is being done by LlamaParse in the backend. **

LlamaParse is an Open Source tool that allows users to extract contents of each page and creates a MarkDown md file as an output.

$ cat agentic.js
import LlamaCloud from "@llamaindex/llama-cloud";
import fs from "fs";
import path from "path";
import { fileURLToPath } from 'url';


// Define __dirname for ES module scope
const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

// Ensure the output folder exists
const outputDir = path.join(__dirname, 'output');
if (!fs.existsSync(outputDir)) {
    fs.mkdirSync(outputDir, { recursive: true });
}



const client = new LlamaCloud({
  apiKey: process.env.LLAMA_CLOUD_API_KEY || "llx",
});

async function processPdf(filePath) {
  try {
    console.log(`Uploading ${filePath}...`);
    const fileObj = await client.files.create({
      file: fs.createReadStream(filePath),
      purpose: "parse",
    });

    console.log(`Parsing ${filePath} (All pages)...`);
    const result = await client.parsing.parse({
      file_id: fileObj.id,
      tier: "agentic",
      version: "latest",
      expand: ["markdown_full", "text_full"],
    });

    const baseName = path.basename(filePath, path.extname(filePath));
// Write the files inside the output folder
fs.writeFileSync(path.join(outputDir, `${baseName}_output.md`), result.markdown_full ?? "");
fs.writeFileSync(path.join(outputDir, `${baseName}_output.txt`), result.text_full ?? "");

    console.log(`✓ Completed ${filePath}`);
  } catch (error) {
    console.error(`✗ Failed processing ${filePath}:`, error);
  }
}

async function main() {
  const pdfFiles = [
    "input/1.pdf",
    "input/2.pdf",
    "input/3.pdf",
    "input/4.pdf",
    "input/5.pdf",
    "input/6.pdf",
    "input/7.pdf",
    "input/8.pdf",
    "input/9.pdf"
  ];

  await Promise.all(pdfFiles.map((file) => processPdf(file)));
  console.log("All files processed successfully.");
}

main().catch(console.error);
```

import os
import re
import sqlite3

# Connect/Create SQLite Database
conn = sqlite3.connect('voters.db')
cursor = conn.cursor()

# Create voters table
cursor.execute('''
CREATE TABLE IF NOT EXISTS voters (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    file_name TEXT,
    polling_station TEXT,
    voter_name TEXT,
    relation_type TEXT,
    relation_name TEXT,
    age INTEGER,
    gender TEXT,
    house_number TEXT
)
''')

output_dir = 'output'

for filename in os.listdir(output_dir):
    if filename.endswith('.md'):
        filepath = os.path.join(output_dir, filename)
        with open(filepath, 'r', encoding='utf-8') as f:
            content = f.read()

        # Extract Polling Station
        ps_match = re.search(r'Polling\s*Station\s*[:\-]?\s*([^\n<]+)', content, re.IGNORECASE)
        polling_station = ps_match.group(1).strip() if ps_match else "Unknown"

        # Regex patterns to extract individual voter records
        # Customize these patterns based on how voter blocks are written in your markdown
        names = re.findall(r'Name\s*[:\-]\s*([^\n<]+)', content, re.IGNORECASE)
        ages = re.findall(r'Age\s*[:\-]\s*(\d+)', content, re.IGNORECASE)
        genders = re.findall(r'Gender\s*[:\-]\s*([^\n<]+)', content, re.IGNORECASE)
        relations = re.findall(r'(Husband|Father|Mother|Wife)\'?s?\s*Name\s*[:\-]\s*([^\n<]+)', content, re.IGNORECASE)

        # Loop through extracted entries and insert into DB
        for i in range(len(names)):
            name = names[i].strip() if i < len(names) else None
            age = int(ages[i]) if i < len(ages) else None
            gender = genders[i].strip() if i < len(genders) else None
            rel_type = relations[i][0] if i < len(relations) else None
            rel_name = relations[i][1].strip() if i < len(relations) else None

            cursor.execute('''
            INSERT INTO voters (file_name, polling_station, voter_name, relation_type, relation_name, age, gender)
            VALUES (?, ?, ?, ?, ?, ?, ?)
            ''', (filename, polling_station, name, rel_type, rel_name, age, gender))

conn.commit()
conn.close()
print("Successfully created voters.db!")

Shashis-MacBook-Air:parse shashikiran$ cat agent.py 
from langchain_community.agent_toolkits import create_sql_agent
from langchain_community.utilities import SQLDatabase
from langchain_ollama import ChatOllama

# Connect to local Gemma 3 model via Ollama
llm = ChatOllama(model="gemma3:4b", temperature=0)

# Connect to the local voter SQLite database
db = SQLDatabase.from_uri("sqlite:///voters.db")

# Create the SQL Agent
agent_executor = create_sql_agent(
    llm=llm,
    db=db,
    agent_type="zero-shot-react-description",
    verbose=True
)

print("=" * 50)
print("Voter AI Assistant Ready! (Type 'exit' or 'quit' to stop)")
print("=" * 50)

# Continuous interactive loop
while True:
    try:
        user_query = input("\nAsk a question: ").strip()
        
        # Check if user wants to exit
        if user_query.lower() in ["exit", "quit", "q", "bye"]:
            print("Exiting assistant. Goodbye!")
            break
            
        if not user_query:
            continue

        print("\nThinking...")
        response = agent_executor.invoke({"input": user_query})
        print(f"\nAnswer: {response['output']}\n")

    except KeyboardInterrupt:
        print("\nExiting assistant. Goodbye!")
        break
    except Exception as e:
        print(f"\nAn error occurred: {e}\nPlease try asking again.")

```
